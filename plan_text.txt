4b. Pipe capacity — Manning's equation (Engine Dev)
# engine/capacity.py
import math
def pipe_capacity(diameter_m, length_m, slope, roughness=0.013):
 r = diameter_m / 2
 A = math.pi * r**2
 P = 2 * math.pi * r
 R_h = A / P
 slope = max(slope, 0.001) # avoid zero slope
return (1/roughness) * A * (R_h ** (2/3)) * math.sqrt(slope) # m³/s
4c. Surcharge → street depth (Engine Dev)
# engine/surcharge.py
def simulate(G, rain_mm_hr, minutes=30):
 depth_cm = {n: 0.0 for n in G.nodes}
for n in G.nodes:
 area = G.nodes[n].get("catch_area", 2000) # m² road draining here
 inflow = rain_mm_hr * area * 1e-3 / 3.6 # m³/s
# distribute to downstream pipes by slope capacity
for nbr in G.neighbors(n):
 e = G[n][nbr]
 q_cap = pipe_capacity(e["diameter"], e["length"], e["slope"])
 taken = min(inflow, q_cap)
 inflow -= taken
 excess = max(0.0, inflow) # pipes couldn't take it
 depth_cm[n] = excess * (minutes*60) / (area * 100)
return depth_cm
4d. Safe ambulance route (Routing Dev)
# routing/safe_route.py
import networkx as nx
def safe_route(road_G, depth_cm, source, target, threshold_cm=15):
 G = road_G.copy()
for n, d in depth_cm.items():
if d > threshold_cm:
 G.remove_node(n) # block flooded roads
return nx.shortest_path(G, source, target, weight="length")
4e. Choke simulation (Routing Dev)
# routing/choke.py
def choke(G, node):
 G.nodes[node]["blocked"] = True # zero its pipes' capacity
for nbr in G.neighbors(node):
 G[node][nbr]["diameter"] = 0.0

10. Jury Q&A — 3 must-have answers (memorise)
"Pipe data kahan se laoge?" → "Real underground sewer data isn't public in India. We auto-generate a 
synthetic proxy drainage network from OpenStreetMap roads + DEM slope using the Pysewer method —
validated in recent hydrology research for data-scarce cities. Topology is realistic; pipe diameters are
assumed."
"Radar/DEM files are huge — won't your API crash?" → "We pre-download a small ~5 km² DEM tile + a
cached Open-Meteo JSON. The demo runs fully offline; live API is a bonus."
"SWMM is slow — how is this real-time?" → "We replaced SWMM with a lightweight Manning-capacity
check in NetworkX — runs in ~2 seconds, sufficient for nowcasting. SWMM can be added later for
production."
